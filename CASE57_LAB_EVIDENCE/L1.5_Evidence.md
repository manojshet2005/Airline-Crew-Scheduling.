import math


def validate_duty(
    hours_today: int | float,
    hours_last_7_days: int | float,
    rest_hours: int | float
) -> tuple[bool, str]:
    """
    Validate airline crew duty using configurable sample rules.

    Sample rules:
    1. Daily duty must not exceed 10 hours.
    2. Rest between duties must be at least 10 hours.
    3. Rolling seven-day duty must not exceed 60 hours.
    Equality at the limits is allowed.

    Raises:
        ValueError: If any input is not an int or float, is a bool,
            is non-finite, or is negative.

    Returns:
        tuple[bool, str]: Validation result and a concise reason.
    """
    values = (hours_today, hours_last_7_days, rest_hours)

    # Validate all inputs before checking duty rules.
    for value in values:
        if type(value) not in (int, float):
            raise ValueError("Inputs must be finite, non-negative numbers.")
        if isinstance(value, float) and not math.isfinite(value):
            raise ValueError("Inputs must be finite, non-negative numbers.")
        if value < 0:
            raise ValueError("Inputs must be finite, non-negative numbers.")

    # Check rules in the required priority order.
    if hours_today > 10:
        return False, "Daily duty exceeds 10 hours."

    if rest_hours < 10:
        return False, "Rest is below 10 hours."

    if hours_last_7_days > 60:
        return False, "Rolling 7-day duty exceeds 60 hours."

    return True, "All sample duty rules satisfied."
